import { describe, it, before, afterEach, mock } from 'node:test';
import assert from 'node:assert/strict';
import { Test, TestingModule } from '@nestjs/testing';
import { NotFoundException } from '@nestjs/common';
import { {{EntityName}}Service } from '../services/{{FileName}}.service';

describe('{{EntityName}}Service', () => {
  let service: {{EntityName}}Service;

  const mockRepository = {
    find: mock.fn(),
    findOneBy: mock.fn(),
    create: mock.fn(),
    save: mock.fn(),
    update: mock.fn(),
    delete: mock.fn(),
  };

  before(async () => {
    const module: TestingModule = await Test.createTestingModule({
      providers: [
        {{EntityName}}Service,
        {
          provide: '{{EntityName}}',
          useValue: mockRepository,
        },
      ],
    }).compile();

    service = module.get<{{EntityName}}Service>({{EntityName}}Service);
  });

  afterEach(() => {
    mockRepository.find.mock.resetCalls();
    mockRepository.findOneBy.mock.resetCalls();
    mockRepository.create.mock.resetCalls();
    mockRepository.save.mock.resetCalls();
    mockRepository.update.mock.resetCalls();
    mockRepository.delete.mock.resetCalls();
  });

  describe('findAll', () => {
    it('should return paginated results', async () => {
      const expected = [{ "{{PrimaryKeyField}}": 1 }];
      mockRepository.find.mock.mockImplementation(() => Promise.resolve(expected));

      const result = await service.findAll({ skip: 0, limit: 10 });

      assert.deepStrictEqual(result, expected);
      assert.equal(mockRepository.find.mock.callCount(), 1);
    });

    it('should default limit to 10 when not provided', async () => {
      mockRepository.find.mock.mockImplementation(() => Promise.resolve([]));

      await service.findAll({ skip: 0 });

      const callArgs = mockRepository.find.mock.calls[0].arguments[0];
      assert.equal(callArgs.take, 10);
    });
  });

  describe('findOne', () => {
    it('should return entity by {{PrimaryKeyField}}', async () => {
      const expected = { "{{PrimaryKeyField}}": 1 };
      mockRepository.findOneBy.mock.mockImplementation(() => Promise.resolve(expected));

      const result = await service.findOne(1);

      assert.deepStrictEqual(result, expected);
    });

    it('should throw NotFoundException when entity not found', async () => {
      mockRepository.findOneBy.mock.mockImplementation(() => Promise.resolve(null));

      await assert.rejects(
        () => service.findOne(999),
        (err) => {
          assert.ok(err instanceof NotFoundException);
          return true;
        },
      );
    });
  });

  describe('create', () => {
    it('should create and return new entity', async () => {
      const dto = {};
      const created = { "{{PrimaryKeyField}}": 1, ...dto };
      mockRepository.create.mock.mockImplementation(() => created);
      mockRepository.save.mock.mockImplementation(() => Promise.resolve(created));

      const result = await service.create(dto as any);

      assert.deepStrictEqual(result, created);
      assert.equal(mockRepository.create.mock.callCount(), 1);
      assert.equal(mockRepository.save.mock.callCount(), 1);
    });
  });

  describe('update', () => {
    it('should update and return entity', async () => {
      const dto = {};
      const updated = { "{{PrimaryKeyField}}": 1, ...dto };
      mockRepository.update.mock.mockImplementation(() => Promise.resolve({ affected: 1 }));
      mockRepository.findOneBy.mock.mockImplementation(() => Promise.resolve(updated));

      const result = await service.update(1, dto as any);

      assert.deepStrictEqual(result, updated);
    });

    it('should throw NotFoundException when entity not found', async () => {
      mockRepository.update.mock.mockImplementation(() => Promise.resolve({ affected: 0 }));

      await assert.rejects(
        () => service.update(999, {} as any),
        (err) => {
          assert.ok(err instanceof NotFoundException);
          return true;
        },
      );
    });
  });

  describe('remove', () => {
    it('should delete and return result', async () => {
      const deleteResult = { affected: 1 };
      mockRepository.delete.mock.mockImplementation(() => Promise.resolve(deleteResult));

      const result = await service.remove(1);

      assert.deepStrictEqual(result, deleteResult);
    });

    it('should throw NotFoundException when entity not found', async () => {
      mockRepository.delete.mock.mockImplementation(() => Promise.resolve({ affected: 0 }));

      await assert.rejects(
        () => service.remove(999),
        (err) => {
          assert.ok(err instanceof NotFoundException);
          return true;
        },
      );
    });
  });
});
